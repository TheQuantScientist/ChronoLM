import sys
sys.path.insert(0, '.')
from data.dependencies.tsdm.tasks.P12 import Physionet2012

task = Physionet2012()
train, val, test = task.splits

print("=== TRAIN ===")
for i, part in enumerate(train):
    print(f"  Part [{i}]: type={type(part)}")
    if hasattr(part, 'shape'):
        print(f"    shape={part.shape}, dtype={part.dtype}")
        print(f"    min={part.min():.4f}, max={part.max():.4f}, mean={part.mean():.4f}")
    elif hasattr(part, '__len__'):
        print(f"    len={len(part)}")
        if hasattr(part, 'columns'):
            print(f"    columns={list(part.columns)[:10]}")
            print(f"    head:\n{part.head()}")
        elif hasattr(part, 'keys'):
            keys = list(part.keys())[:5]
            print(f"    keys (first 5): {keys}")
            for k in keys[:2]:
                v = part[k]
                print(f"    [{k}]: type={type(v)}, shape={v.shape if hasattr(v, 'shape') else len(v)}")

print("\n=== TEST ===")
for i, part in enumerate(test):
    print(f"  Part [{i}]: type={type(part)}")
    if hasattr(part, 'shape'):
        print(f"    shape={part.shape}, dtype={part.dtype}")
        print(f"    min={part.min():.4f}, max={part.max():.4f}, mean={part.mean():.4f}")
    elif hasattr(part, '__len__'):
        print(f"    len={len(part)}")
